Supabase Schema DDL

Paste this SQL script in your Supabase SQL Editor to prepare your tables:

-- Create profiles table for user accounts
CREATE TABLE IF NOT EXISTS public.profiles (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  username text NOT NULL,
  email text UNIQUE NOT NULL,
  password_hash text NOT NULL,
  household text,
  preferences jsonb DEFAULT '{}'::jsonb,
  created_at timestamp with time zone DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- Enable RLS for profiles
ALTER TABLE public.profiles ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow public access for dev" ON public.profiles FOR ALL USING (true) WITH CHECK (true);

-- Create food_items table for refrigerator items
CREATE TABLE IF NOT EXISTS public.food_items (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id text NOT NULL,
  name text NOT NULL,
  category text NOT NULL,
  quantity numeric NOT NULL,
  quantity_unit text DEFAULT 'pieces',
  storage_location text DEFAULT 'Shelf',
  expiry_date date,
  has_expiry boolean DEFAULT true,
  days_left integer,
  priority_score integer DEFAULT 0,
  created_at timestamp with time zone DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- Enable RLS for food_items
ALTER TABLE public.food_items ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow public access for dev" ON public.food_items FOR ALL USING (true) WITH CHECK (true);

-- Create notifications table
CREATE TABLE IF NOT EXISTS public.notifications (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id text NOT NULL,
  title text NOT NULL,
  message text NOT NULL,
  type text DEFAULT 'info',
  is_read boolean DEFAULT false,
  created_at timestamp with time zone DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- Enable RLS for notifications
ALTER TABLE public.notifications ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow public access for dev" ON public.notifications FOR ALL USING (true) WITH CHECK (true);
